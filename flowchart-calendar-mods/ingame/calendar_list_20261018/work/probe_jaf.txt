// the k-th list cell under (month, lane) in list order, or -1
int KR_PRNth(int month, int lane, int k) {
    int t0 = 0;
    int t1 = 7;
    if (month >= 0) { t0 = month; t1 = month; }
    int c = 0;
    for (int t = t0; t <= t1; t++) {
        for (int i = KR_CTabFirst(t); i < KR_CTabEnd(t); i++) {
            if (!KR_LVMatch(i, lane)) continue;
            if (c == k) return i;
            c++;
        }
    }
    return -1;
}
// the expected count from the board's own counter (KR_XVCount totals of the rows shown)
int KR_PRExpect(int month, int lane) {
    int t0 = 0;
    int t1 = 7;
    if (month >= 0) { t0 = month; t1 = month; }
    int n = 0;
    for (int ln = 0; ln <= 9; ln++) {
        if (ln == 1) continue;
        if (lane >= 0 && ln != lane) continue;
        if (!KR_LVLaneOn(ln)) continue;
        int pk = KR_XVCount(ln, t0, t1, false);
        n = n + pk % 10000;
    }
    return n;
}

    if (frames > 3000) { KR_PRLog("STUCK dialog stage=" + stage); KR_PRShot("lv_stuck.bmp"); system.Exit(5); return; }

        if (stage == "home0" && frames < 600) return;
        KR_PRLog("UNEXPECTED DIALOG stage=" + stage + " frames=" + ("%d" % frames));
        KR_PRShot("lv_unexpected_dialog.bmp");
        system.Exit(7);
        return;
    }

        KR_PRCheck("home panel shown with the calendar button", KR_PRRate() == 1 && KR_PRShown(act, "KR_HR_CalBox") == 1);

        KR_PRSet("CFrames", "0");
        KR_T8HomeClick(KR_PRP(act, "KR_HR_CalBox"), 80, @@CAL_Y@@ - KR_HVUp() + 12);
        return;
    }
    if (stage == "home1") {
        if (frames < 40) return;
        KR_PRCheck("closing the board returns to the home menu", KR_PRRate() == 1 && !PartsEngine.IsExist(KR_XVP(0)) && !PartsEngine.IsExist(KR_CRoot()));
        if (KR_PRMode() != "nov") { KR_PREndAll(); return; }

        KR_T8HomeClick(KR_PRP(act, "KR_HR_Box2"), 80, @@ROW_Y0@@ + 20);

    KR_PRShot("lv_unexpected.bmp");

}
bool KR_PRRowsMetOK() {
    string rs = KR_XVGet("rows");
    int n = rs.ToInt();
    if (KR_XVGet("row0") != "0") return false;
    if (KR_XVGet("row%d" % (n - 1)) != "9") return false;
    int k = 1;
    for (int p = 2; p <= 8; p++) {
        if (!KR_XVMet(p)) continue;
        string v = KR_XVGet("row%d" % k);
        if (v.ToInt() != p) return false;
        k++;
    }
    return k == n - 1;
}
void KR_ProbeTitle(string stage, int f) {
    if (stage == "t_b") {
        if (f == 30) {
            KR_PRCheck("title: the menubar request opens the board", KR_PRBoard() && !PartsEngine.IsExist(KR_XVP(3)) && KR_PRRowsMetOK());
            KR_PRShot("lv_title_board.bmp");
        }
        if (f == 32) KR_PRGo("t_l", KR_XVP(23));
        return;
    }
    if (stage == "t_l") {
        if (f == 30) {
            int n = KR_PRExpect(3, -1);
            KR_PRLog("TITLE LIST n=" + ("%d" % n) + " foot=" + KR_PRTxt(KR_LVP(71)));
            KR_PRCheck("title: the 7월 head opens every row of 7월, no 이 세이브 button", KR_PRList() && KR_PRHead(3, -1) && KR_PRFoot(n) && !PartsEngine.IsExist(KR_LVP(3)));
            KR_PRShot("lv_title_list.bmp");
        }
        if (f == 32) KR_PRGo("t_m", KR_LVP(54));
        return;
    }
    if (stage == "t_m") {
        if (f == 30) {
            KR_PRLog("MONTH head=" + KR_PRTxt(KR_CPart(1)));
            KR_PRCheck("title: 달 화면으로 opens the month view on 7월 (list parts released)",
                       KR_CGet("Open") == "1" && KR_LVGet("Open") != "1" && KR_PRHas(KR_CPart(1), "7월") && !PartsEngine.IsExist(KR_LVP(1)));
            KR_PRShot("lv_title_month.bmp");
        }
        if (f == 32) KR_PRGo("t_l2", 0);
        return;
    }
    if (stage == "t_l2") {
        if (f == 30) KR_PRCheck("title: closing the month view returns to the list (same filter)", KR_PRList() && KR_PRHead(3, -1));
        if (f == 32) KR_PRGo("t_b2", KR_LVP(7));
        return;
    }
    if (stage == "t_b2") {
        if (f == 30) {
            KR_PRCheck("title: 닫기 on the list returns to the board", KR_PRBoard());
            KR_PRShot("lv_title_board_back.bmp");
            KR_PREndAll();
        }
        return;
    }
    KR_PRLog("CHART TICK in stage " + stage);
    KR_PRShot("lv_unexpected.bmp");
    system.Exit(6);
}
void KR_ProbeChartTick() {
    string stage = KR_PRStage();

    else if (KR_PRMode() == "may") KR_ProbeMay(stage, f);

}
